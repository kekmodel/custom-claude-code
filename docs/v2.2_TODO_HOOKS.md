```python
#    )
# 2. Hook에서 보존할 메시지 마킹 가능
# 3. 메시지 압축 수행 (RemoveMessage 사용)
# 4. 압축된 메시지로 교체
```

### 트리거 시점
- 사용자 입력 처리 직전
- 메시지를 messages 리스트에 추가하기 전
- 메시지 수 또는 토큰 수가 임계값을 초과했을 때

### 구현 시 고려사항

1. **임계값 설정**
   - 메시지 수: 100개 이상
   - 토큰 수: 100,000 토큰 이상 (권장)

2. **Hook input_data 구조**
   ```python
   {
       "messages": list[BaseMessage],  # 현재 메시지 목록
       "threshold": int,                # 초과된 임계값
       "current_token_count": int,      # 현재 총 토큰 수 (선택적)
   }
   ```

3. **Hook 반환값 구조**
   ```python
   {
       "hookSpecificOutput": {
           "hookEventName": "PreCompact",
           "preserveMessageIds": [str],  # 보존할 메시지 ID 목록 (선택적)
           "compactionStrategy": str,    # "summary" | "selective" | "custom"
       }
   }
   ```

4. **압축 전략**
   - **summary**: LLM으로 대화 요약 생성 후 오래된 메시지 제거
   - **selective**: 중요한 메시지만 보존 (에러, 경고, 최근 N개)
   - **custom**: Hook에서 제공한 preserveMessageIds만 보존

### 참고 구현 (v2.1 스타일)

```python
from langchain_core.messages import RemoveMessage

# 1. 토큰 수 계산 (간단한 근사치)
total_tokens = sum(len(msg.content) // 4 for msg in messages)

# 2. 임계값 체크
if total_tokens > 100000:
    # 3. PreCompact Hook 트리거
    hook_result = await trigger_hook(
        event="PreCompact",
        input_data={
            "messages": messages,
            "threshold": 100000,
            "current_token_count": total_tokens
        },
        tool_use_id=None,
        context=HookContext(session_id="main", turn_count=len(messages))
    )

    # 4. Hook에서 보존할 메시지 확인
    hook_output = hook_result.get("hookSpecificOutput", {})
    preserve_ids = hook_output.get("preserveMessageIds", [])

    # 5. 메시지 압축
    # Option A: 요약 생성 (LLM 사용)
    if hook_output.get("compactionStrategy") == "summary":
        summary = await generate_summary(messages)
        messages_to_keep = [messages[0], HumanMessage(content=summary)] + messages[-10:]

    # Option B: 선택적 보존
    else:
        messages_to_keep = [msg for msg in messages if msg.id in preserve_ids or should_preserve(msg)]

    # 6. 메시지 교체
    messages.clear()
    messages.extend(messages_to_keep)
```

```python
async def stop_hook_example(input_data, tool_use_id, context):
    """Stop Hook: 세션 저장 및 통계 기록"""
    reason = input_data.get("reason")
    message_count = input_data.get("message_count", 0)

    # 1. 세션 데이터 저장
    session_data = {
        "session_id": context.session_id,
        "message_count": message_count,
        "stop_reason": reason,
        "timestamp": datetime.now().isoformat()
    }

    # 2. 파일에 저장
    with open(f".sessions/{context.session_id}.json", "w") as f:
        json.dump(session_data, f, indent=2)

    # 3. 통계 업데이트
    update_statistics(context.session_id, message_count)

    # 4. 정리 작업 로그
    print(f"[Hook] Session saved: {context.session_id}")
    print(f"[Hook] Total messages: {message_count}")

    return {
        "hookSpecificOutput": {
            "hookEventName": "Stop",
            "cleanupCompleted": True,
            "savedData": {
                "session_id": context.session_id,
                "file_path": f".sessions/{context.session_id}.json"
            }
        }
    }
```
